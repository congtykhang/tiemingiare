function doPost(e) { var lock = LockService.getScriptLock(); try { lock.waitLock(10000); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: 'Hệ thống đang bận, vui lòng thử lại sau 2 giây.' })).setMimeType(ContentService.MimeType.JSON); } try { var payload = JSON.parse(e.postData.contents); var chieuDai = payload.dai || 0; var chieuRong = payload.rong || 0; var soLuong = payload.soluong || 0; var chatLieu = payload.chatlieu || "Decal giấy"; var canMang = payload.canmang || "Không"; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("TÍNH GIÁ TỔNG HỢP"); // 1. GHI GỘP 1 LẦN VÀO CẢ DẢI Ô B4:B8 sheet.getRange("B4:B8").setValues([ [chieuDai], [chieuRong], [soLuong], [chatLieu], [canMang] ]); SpreadsheetApp.flush(); // 2. ĐỌC GỘP 1 LẦN DẢI Ô B15:B23 CHO SỐ LƯỢNG GỐC var resRange1 = sheet.getRange("B15:B23").getValues(); var fastPrice1 = resRange1[0][0]; var offsetPrice1 = resRange1[8][0]; // 3. TÍNH & LẤY SẴN TRỌN BỘ GIÁ CHO MỐC CHẴN NGÀN TIẾP THEO var nextQty = (soLuong < 1000) ? 1000 : Math.ceil(soLuong / 1000) * 1000; var fastPrice2 = 0; var offsetPrice2 = 0; if (soLuong !== nextQty) { sheet.getRange("B6").setValue(nextQty); SpreadsheetApp.flush(); var resRange2 = sheet.getRange("B15:B23").getValues(); fastPrice2 = resRange2[0][0]; offsetPrice2 = resRange2[8][0]; } var result = { success: true, originalQty: soLuong, nextQty: nextQty, tongTienFast: (fastPrice1 && !isNaN(fastPrice1) && fastPrice1 > 0) ? Math.round(Number(fastPrice1)) : 0, tongTienOffset: (offsetPrice1 && !isNaN(offsetPrice1) && offsetPrice1 > 0) ? Math.round(Number(offsetPrice1)) : 0, tongTienFastNext: (fastPrice2 && !isNaN(fastPrice2) && fastPrice2 > 0) ? Math.round(Number(fastPrice2)) : 0, tongTienOffsetNext: (offsetPrice2 && !isNaN(offsetPrice2) && offsetPrice2 > 0) ? Math.round(Number(offsetPrice2)) : 0 }; lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON); } catch (error) { lock.releaseLock(); return ContentService.createTextOutput(JSON.stringify({ success: false, error: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
